cmake_minimum_required(VERSION 3.20)
project(discordnx2_voice_deps LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3.0 REQUIRED)
file(WRITE "${CMAKE_BINARY_DIR}/namespace.h" "#pragma once\n#define MLS_NAMESPACE mlspp\n")
set(MLS "${DNX_VOICE_SOURCES}/mls")
set(DAVE "${DNX_VOICE_SOURCES}/dave/cpp")
file(GLOB MLS_SRC "${MLS}/src/*.cpp" "${MLS}/lib/bytes/src/*.cpp" "${MLS}/lib/tls_syntax/src/*.cpp" "${MLS}/lib/hpke/src/*.cpp")
file(GLOB_RECURSE DAVE_SRC "${DAVE}/src/*.cpp")
list(FILTER DAVE_SRC EXCLUDE REGEX "bindings_wasm|boringssl_cryptor|persisted_key|_win\\.cpp|_apple\\.cpp")
list(APPEND DAVE_SRC "${DAVE}/src/mls/persisted_key_pair_null.cpp")
add_library(dnx_dave STATIC ${MLS_SRC} ${DAVE_SRC})
target_include_directories(dnx_dave PUBLIC "${DAVE}/includes" PRIVATE "${MLS}/include" "${DAVE}/src" "${MLS}/lib/bytes/include" "${MLS}/lib/tls_syntax/include" "${MLS}/lib/hpke/include" "${MLS}/third_party" "${CMAKE_BINARY_DIR}" "${DNX_VOICE_SOURCES}/json")
target_compile_definitions(dnx_dave PUBLIC DAVE_STATIC PRIVATE WITH_OPENSSL3 DISABLE_GREASE)
target_compile_options(dnx_dave PRIVATE -O2 -Wno-deprecated-declarations)
target_link_libraries(dnx_dave PUBLIC OpenSSL::Crypto)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OPUS_ENABLE_INTRINSICS OFF CACHE BOOL "" FORCE)
set(OPUS_STACK_PROTECTOR OFF CACHE BOOL "" FORCE)
set(OPUS_FORTIFY_SOURCE OFF CACHE BOOL "" FORCE)
add_subdirectory("${DNX_VOICE_SOURCES}/opus" opus)
install(TARGETS dnx_dave opus ARCHIVE DESTINATION lib)
install(DIRECTORY "${DAVE}/includes/" DESTINATION include)
install(FILES "${DNX_VOICE_SOURCES}/opus/include/opus.h" "${DNX_VOICE_SOURCES}/opus/include/opus_types.h" "${DNX_VOICE_SOURCES}/opus/include/opus_defines.h" DESTINATION include/opus)
